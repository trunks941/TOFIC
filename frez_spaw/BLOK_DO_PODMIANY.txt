NOP
'FREZ 12MM TOOL 17
'START OSI FREZA: PROMIEN 3MM
'I073: KROK SREDNICY W 0.001MM
'D120: PROMIEN DOCELOWY
'D121: KROK PROMIENIA NA SZTUKE
'D103-D105 Z TROJNIK_TEST
*PARAMETRY
IFTHENEXP I035=30
	 NOP
ELSE
	 PAUSE
	 JUMP *PARAMETRY
ENDIF
IFTHENEXP I120<1
	 PAUSE
	 JUMP *PARAMETRY
ENDIF
IFTHENEXP I120=1
	'NOWA SZTUKA: SREDNICE
	 IFTHENEXP I070<=0
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
	 IFTHENEXP I074<0
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
	 SET D073  EXPRESS I071 * 100
	 SET D075  EXPRESS I074 * 100
	 MUL D075 2
	 SET D072  EXPRESS I070 * 100
	 SET D079 D072
	 SET D120  EXPRESS D073 - D075 - D072
	 DIV D120 2
	 IFTHENEXP D120<3000
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
	'KROK SREDNICY: DODATNI PARZYSTY
	 SET D121 I073
	 IFTHENEXP D121<2
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
	 SET D066 D121
	 DIV D066 2
	 MUL D066 2
	 IFTHENEXP D066=D121
		 DIV D121 2
	 ELSE
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
	 SET B030 0
	 DOUT OT#(6) OFF
	 SET D089  EXPRESS I089 * 1000
ELSE
	 IFTHENEXP D121<=0
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
	 IFTHENEXP D120<3000
		 PAUSE
		 JUMP *PARAMETRY
	 ENDIF
ENDIF
'PROMIEN Z NUMERU OBIEGU
SET D074  EXPRESS I120 - 1
MUL D074 D121
ADD D074 3000
'OSTATNI KROK DO GRANICY
IFTHENEXP D074>D120
	 SET D074 D120
ENDIF
'PUNKTY DLA TOOL 17
FOR I060 START= 10 TO 13 STEP= 1
	 SETE P[I060] TL#(17)
NEXT I060
